1. Express Middleware (Axiom26 Backend — src/middleware/telemetry.ts) 
// Drop into Axiom26 Express app before routes to expose /api/admin/ops-telemetry
import type { Request, Response, NextFunction } from "express";

const routeStats = new Map<string, { count: number; errors: number; latencies: number[] }>();

export function opsTelemetryMiddleware(req: Request, res: Response, next: NextFunction) {
  const start = performance.now();
  res.on("finish", () => {
    const key = `${req.method} ${req.route?.path || req.path}`;
    const entry = routeStats.get(key) || { count: 0, errors: 0, latencies: [] };
    entry.count += 1;
    if (res.statusCode >= 400) entry.errors += 1;
    entry.latencies.push(Math.round(performance.now() - start));
    if (entry.latencies.length > 200) entry.latencies.shift();
    routeStats.set(key, entry);
  });
  next();
}

export function getOpsSnapshot(io: any, prismaPoolActive: number) {
  return {
    timestamp: new Date().toISOString(),
    memoryMb: Math.round(process.memoryUsage().heapUsed / 1024 / 1024),
    socketConnections: io?.engine?.clientsCount ?? 0,
    dbPoolActive: prismaPoolActive,
    endpoints: Array.from(routeStats.entries()).map(([route, s]) => {
      const sorted = [...s.latencies].sort((a, b) => a - b);
      return {
        route,
        rpm: s.count,
        p50: sorted[Math.floor(sorted.length * 0.5)] || 0,
        p95: sorted[Math.floor(sorted.length * 0.95)] || 0,
        errRate: s.count ? Number(((s.errors / s.count) * 100).toFixed(2)) : 0
      };
    })
  };
}

// Attach to src/services/api.js & main.jsx to report client cache hit-rate & LCP
export const clientTelemetry = { cacheHits: 0, cacheMisses: 0, deduped: 0, lcpMs: null };

if (typeof PerformanceObserver !== "undefined") {
  new PerformanceObserver((list) => {
    const entries = list.getEntries();
    const last = entries[entries.length - 1];
    if (last) clientTelemetry.lcpMs = Math.round(last.startTime);
  }).observe({ type: "largest-contentful-paint", buffered: true });
